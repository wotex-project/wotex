cmake_minimum_required(VERSION 3.16)
project(wotex_thread_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WOTEX_NATIVE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
set(WOTEX_NATIVE_TEST_SOURCE "" CACHE PATH "Explicit source directory containing native tests")

set(json_header "${CMAKE_BINARY_DIR}/include/nlohmann/json.hpp")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/nlohmann")
set(WOTEX_JSON_HEADER "" CACHE FILEPATH "Explicit pinned JSON header for the Mix native build")
if(WOTEX_JSON_HEADER)
  file(SHA256 "${WOTEX_JSON_HEADER}" supplied_json_sha256)
  if(NOT supplied_json_sha256 STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Pinned JSON header digest mismatch")
  endif()
  configure_file("${WOTEX_JSON_HEADER}" "${json_header}" COPYONLY)
else()
  file(DOWNLOAD
    "https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp"
    "${json_header}"
    EXPECTED_HASH SHA256=9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6
    TLS_VERIFY ON TIMEOUT 120)
endif()

add_library(wotex-thread-protocol INTERFACE)
target_include_directories(wotex-thread-protocol INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(wotex-thread-protocol SYSTEM INTERFACE "${CMAKE_BINARY_DIR}/include")
target_compile_options(wotex-thread-protocol INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(WOTEX_NATIVE_SANITIZERS)
  target_compile_definitions(wotex-thread-protocol INTERFACE WOTEX_NATIVE_SANITIZERS=1)
  target_compile_options(wotex-thread-protocol INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(wotex-thread-protocol INTERFACE -fsanitize=address,undefined)
endif()

if(WOTEX_NATIVE_TEST_SOURCE)
  enable_testing()
  add_executable(wotex-thread-protocol-test "${WOTEX_NATIVE_TEST_SOURCE}/protocol_test.cpp")
  target_link_libraries(wotex-thread-protocol-test PRIVATE wotex-thread-protocol)

  add_executable(wotex-thread-output-test "${WOTEX_NATIVE_TEST_SOURCE}/output_test.cpp")
  target_link_libraries(wotex-thread-output-test PRIVATE wotex-thread-protocol)
  add_test(NAME WTH-B02-OUTPUT COMMAND wotex-thread-output-test)
  add_executable(wotex-thread-streams-test "${WOTEX_NATIVE_TEST_SOURCE}/streams_test.cpp")
  target_link_libraries(wotex-thread-streams-test PRIVATE wotex-thread-protocol)
  add_test(NAME WTH-S06-STREAMS COMMAND wotex-thread-streams-test)
  add_executable(wotex-thread-flow-test "${WOTEX_NATIVE_TEST_SOURCE}/flow_test.cpp")
  target_link_libraries(wotex-thread-flow-test PRIVATE wotex-thread-protocol)

  target_link_libraries(wotex-thread-contract-driver PRIVATE wotex-thread-protocol)
  set_tests_properties(WTH-C07-WTH-V04 WTH-S03-WTH-V04 WTH-B02-OUTPUT WTH-S06-STREAMS

set(WOTEX_OPENTHREAD_SOURCE "" CACHE PATH "Explicit pinned OpenThread source tree")
if(WOTEX_OPENTHREAD_SOURCE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The SDK host requires Linux; pure native tests are portable")
  endif()
  file(SHA256 "${WOTEX_OPENTHREAD_SOURCE}/src/lib/spinel/spinel.c" spinel_sha256)
  if(NOT spinel_sha256 STREQUAL "28841f049ab1c95c8a0b6baa73d7f6579ff2f97f007ef1496751453dfe4ba57b")
    message(FATAL_ERROR "The pinned SDK requires the audited unsigned Spinel shift fix; use mix wotex.native.build")
  endif()
  file(SHA256 "${WOTEX_OPENTHREAD_SOURCE}/src/core/meshcop/meshcop.hpp" discerner_sha256)
  if(NOT discerner_sha256 STREQUAL "4dd6b16b20b3df58756fdfe1519404e9520fcef16d916c9848e983bdf97edfa6")
    message(FATAL_ERROR "The pinned SDK requires the full-width discerner fix; use mix wotex.native.build")
  endif()
  set(OT_PLATFORM posix CACHE STRING "" FORCE)
  set(OT_APP_CLI OFF CACHE BOOL "" FORCE)
  set(OT_APP_NCP OFF CACHE BOOL "" FORCE)
  set(OT_APP_RCP OFF CACHE BOOL "" FORCE)
  set(OT_MTD OFF CACHE BOOL "" FORCE)
  set(OT_RCP OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(OT_COMMISSIONER ON CACHE BOOL "" FORCE)
  set(OT_JOINER ON CACHE BOOL "" FORCE)
  set(OT_LOG_LEVEL NONE CACHE STRING "" FORCE)
  set(OT_POSIX_RCP_HDLC_BUS ON CACHE BOOL "" FORCE)
  set(OT_POSIX_RCP_SPI_BUS ON CACHE BOOL "" FORCE)
  set(OT_PLATFORM_NETIF ON CACHE STRING "" FORCE)
  set(OT_PLATFORM_UDP ON CACHE STRING "" FORCE)
  set(OT_PACKAGE_VERSION "wotex-f34c5e5476829d9205e80b37fccc2bdfe97e1dab" CACHE STRING "" FORCE)
  add_subdirectory("${WOTEX_OPENTHREAD_SOURCE}" "${CMAKE_BINARY_DIR}/sdk")
  target_compile_definitions(ot-config INTERFACE "OPENTHREAD_BUILD_DATETIME=\"pinned-source\"")
  if(WOTEX_NATIVE_SANITIZERS)
    target_compile_options(ot-config INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(ot-config INTERFACE -fsanitize=address,undefined)
    foreach(crypto_target mbedtls mbedx509 mbedcrypto)
      target_compile_options(${crypto_target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    endforeach()
  endif()
  add_executable(wotex-thread-host host.cpp)
  set(host_targets wotex-thread-host)
  if(WOTEX_NATIVE_TEST_SOURCE)
    add_executable(wotex-thread-dataset-seed "${WOTEX_NATIVE_TEST_SOURCE}/dataset_seed.cpp")
    # The production host plus a test-only State callback source for process-flow cases.
    add_executable(wotex-thread-flow-host host.cpp)
    target_compile_definitions(wotex-thread-flow-host PRIVATE WOTEX_THREAD_FLOW_TESTING=1)
    add_executable(wotex-thread-coap-peer "${WOTEX_NATIVE_TEST_SOURCE}/coap_peer.cpp")
    list(APPEND host_targets wotex-thread-dataset-seed wotex-thread-flow-host wotex-thread-coap-peer)
    add_executable(wotex-thread-spinel-test "${WOTEX_NATIVE_TEST_SOURCE}/spinel_test.c")
    target_include_directories(wotex-thread-spinel-test PRIVATE "${WOTEX_OPENTHREAD_SOURCE}/src/lib/spinel")
    target_link_libraries(wotex-thread-spinel-test PRIVATE wotex-thread-protocol openthread-spinel-rcp)
    add_test(NAME WTH-S03-WTH-V04-SPINEL COMMAND wotex-thread-spinel-test)
    set_tests_properties(WTH-S03-WTH-V04-SPINEL PROPERTIES TIMEOUT 30)
  endif()
  foreach(host_target ${host_targets})
    target_include_directories(${host_target} PRIVATE "${WOTEX_OPENTHREAD_SOURCE}/include")
    target_link_libraries(${host_target} PRIVATE wotex-thread-protocol openthread-posix
      openthread-ftd openthread-posix openthread-cli-ftd openthread-hdlc
      openthread-radio-spinel openthread-spinel-rcp ot-config-ftd ot-config)
  endforeach()
endif()
