cmake_minimum_required(VERSION 3.16)
project(wotex-bacnet-fixture LANGUAGES C)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The independent native fixture runs on Linux")
endif()
if(NOT IS_ABSOLUTE "${WOTEX_BACNET_STACK_SOURCE}" OR
   NOT EXISTS "${WOTEX_BACNET_STACK_SOURCE}/src/bacnet/cov.c")
  message(FATAL_ERROR "WOTEX_BACNET_STACK_SOURCE must name the verified SDK source")
endif()

option(WOTEX_BACNET_SANITIZE "Instrument the fixture and linked C stack" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BACNET_STACK_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(BACDL_BIP ON CACHE BOOL "" FORCE)
foreach(option BACDL_BIP6 BACDL_ZIGBEE BACDL_BSC BACDL_MSTP BACDL_ARCNET
               BACDL_ETHERNET BAC_ROUTING BACNET_BACKUP_RESTORE
               BACNET_SEGMENTATION_ENABLED INTRINSIC_REPORTING)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_compile_definitions(MAX_COV_SUBSCRIPTIONS=16 MAX_COV_ADDRESSES=16)
if(WOTEX_BACNET_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory("${WOTEX_BACNET_STACK_SOURCE}" sdk EXCLUDE_FROM_ALL)
get_target_property(sdk_definitions bacnet-stack COMPILE_DEFINITIONS)
list(REMOVE_ITEM sdk_definitions PRINT_ENABLED=1)
set_target_properties(bacnet-stack PROPERTIES COMPILE_DEFINITIONS "${sdk_definitions}")
target_compile_definitions(bacnet-stack PRIVATE PRINT_ENABLED=0)
add_executable(wotex-bacnet-peer peer.c control.c property_peer.c)
target_compile_options(wotex-bacnet-peer PRIVATE -Wall -Wextra -Werror)
target_link_libraries(wotex-bacnet-peer PRIVATE bacnet-stack)
target_link_options(wotex-bacnet-peer PRIVATE -Wl,--wrap=bip_send_pdu)
enable_testing()
add_test(NAME WBA-CP01-control-boundaries COMMAND wotex-bacnet-peer --self-test)
add_test(NAME WBA-CP10-property-boundaries COMMAND wotex-bacnet-peer --property-self-test)
